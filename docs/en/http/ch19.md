## 19.1 Structured Logging

```python
import logging
import requests

# 配置 requests 日志
logging.basicConfig(level=logging.DEBUG)
http_client_logger = logging.getLogger('httpx' or 'requests')
http_client_logger.setLevel(logging.INFO)

# 启用请求日志
import urllib3
urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)

resp = requests.get('https://api.example.com/data')
logging.info(f'Request completed: status={resp.status_code}')
```

## 19.3 Collecting Performance Metrics

```python
import time
import requests

def timed_request(url: str, **kwargs) -> dict:
    start = time.perf_counter()
    resp = requests.get(url, **kwargs)
    elapsed = time.perf_counter() - start
    
    return {
        'url': url,
        'status': resp.status_code,
        'duration_ms': round(elapsed * 1000, 2),
        'size_bytes': len(resp.content)
    }

# 使用
result = timed_request('https://api.example.com/data')
print(f"耗时: {result['duration_ms']}ms")
```

## 19.4 Health Checks

```python
import requests
from typing import Dict, List

def check_endpoints(endpoints: List[str]) -> Dict[str, bool]:
    """检查多个 API 端点的可用性"""
    results = {}
    for url in endpoints:
        try:
            resp = requests.head(url, timeout=5)
            results[url] = resp.status_code < 500
        except requests.exceptions.RequestException:
            results[url] = False
    return results

# 使用
health = check_endpoints([
    'https://api.example.com/health',
    'https://db.example.com/status'
])
```

## 19.5 Graceful Shutdown

```python
import signal
import requests

session = requests.Session()

def graceful_shutdown(signum, frame):
    print('关闭连接池...')
    session.close()
    exit(0)

signal.signal(signal.SIGTERM, graceful_shutdown)
signal.signal(signal.SIGINT, graceful_shutdown)

# 正常处理请求
resp = session.get('https://api.example.com/data')
```
