## 6.1 Conditional Edges Basics

Conditional edges decide the next node based on current state:

```python
from langgraph.graph import StateGraph, MessagesState, START, END

def router(state: MessagesState) -> str:
    """Route based on last message"""
    last_msg = state["messages"][-1]

    if "weather" in last_msg.content:
        return "weather_node"
    elif "search" in last_msg.content:
        return "search_node"
    elif "calculate" in last_msg.content:
        return "calc_node"
    else:
        return "default_node"

graph = StateGraph(MessagesState)
graph.add_node("weather", weather_node)
graph.add_node("search", search_node)
graph.add_node("calc", calc_node)
graph.add_node("default", default_node)

graph.add_edge(START, "default")
graph.add_conditional_edges("default", router, {
    "weather_node": "weather",
    "search_node": "search",
    "calc_node": "calc",
    "default_node": END
})
```

## 6.2 Subgraphs: Modular Design

Subgraphs allow encapsulating complex logic into independent units:

```python
from langgraph.graph import StateGraph, MessagesState, START, END

# Search subgraph
def create_search_subgraph():
    search_state = MessagesState

    def search_node(state):
        return {"messages": [{"role": "assistant", "content": "Search results..."}]}

    sg = StateGraph(search_state)
    sg.add_node("search", search_node)
    sg.add_edge(START, "search")
    sg.add_edge("search", END)
    return sg.compile()

# Analysis subgraph
def create_analysis_subgraph():
    analysis_state = MessagesState

    def analyze_node(state):
        return {"messages": [{"role": "assistant", "content": "Analysis report..."}]}

    sg = StateGraph(analysis_state)
    sg.add_node("analyze", analyze_node)
    sg.add_edge(START, "analyze")
    sg.add_edge("analyze", END)
    return sg.compile()

# Main graph
def create_main_graph():
    main_state = MessagesState

    def route_node(state):
        last_msg = state["messages"][-1]
        if "search" in last_msg.content:
            return "search_subgraph"
        return "analyze_subgraph"

    graph = StateGraph(main_state)
    graph.add_node("route", route_node)
    graph.add_node("search_subgraph", create_search_subgraph())
    graph.add_node("analyze_subgraph", create_analysis_subgraph())

    graph.add_edge(START, "route")
    graph.add_conditional_edges("route", lambda s: "search_subgraph" if "search" in s["messages"][-1].content else "analyze_subgraph", {
        "search_subgraph": END,
        "analyze_subgraph": END
    })

    return graph.compile()
```

## 6.3 Multi-Step Search Agent

```python
from langgraph.graph import StateGraph, MessagesState, START, END
from typing import Annotated, operator

class SearchState(MessagesState):
    queries: Annotated[list, operator.add]
    results: Annotated[list, operator.add]
    need_second_search: bool

def search_node(state: SearchState) -> dict:
    query = state["messages"][-1]["content"]
    # Simulate search
    results = [f"Search result 1: {query}", f"Search result 2: {query}"]
    return {
        "queries": [query],
        "results": results,
        "need_second_search": len(results) < 2
    }

def should_continue(state: SearchState) -> str:
    if state.get("need_second_search"):
        return "refine_search"
    return "summarize"

def refine_search_node(state: SearchState) -> dict:
    # Generate more precise query based on first results
    refined_query = f"{state['queries'][-1]} more detailed"
    return {"queries": [refined_query], "results": [f"Refined result: {refined_query}"]}

def summarize_node(state: SearchState) -> dict:
    all_results = "\n".join(state.get("results", []))
    return {
        "messages": [{
            "role": "assistant",
            "content": f"Summarized search results:\n{all_results}"
        }]
    }

# Build graph
workflow = StateGraph(SearchState)
workflow.add_node("search", search_node)
workflow.add_node("refine", refine_search_node)
workflow.add_node("summarize", summarize_node)

workflow.add_edge(START, "search")
workflow.add_conditional_edges("search", should_continue, {
    "refine_search": "refine",
    "summarize": "summarize"
})
workflow.add_edge("refine", "summarize")
workflow.add_edge("summarize", END)

graph = workflow.compile()
```

- Subgraphs implement modularization and code reuse
- Multi-step search Agent is a typical application of conditional edges
