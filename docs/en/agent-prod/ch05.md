## 5.1 MessagesState 详解

`MessagesState`  is LangGraph  of内置state类型，contains `messages` field：

```python
from langgraph.graph import MessagesState
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage

# messages field结构
messages = [
    HumanMessage(content="你好"),           # user message
    AIMessage(content="你好！有什么可以帮你的？"),  # AI reply
    # inserted when tool call
    AIMessage(
        content="",
        tool_calls=[{"id": "call_123", "name": "get_weather", "args": {"city": "北京"}}]
    ),
    ToolMessage(content="北京今天晴朗，25°C", tool_call_id="call_123")  # tool result
]
```

## 5.2 自定义state

```python
from langgraph.graph import StateGraph, MessagesState, START, END
from typing import TypedDict, Annotated
import operator

class AgentState(MessagesState):
    # use Annotated + operator.add implement message append（non-overwrite）
    documents: Annotated[list, operator.add]  # retrieved document list
    step_count: int  # executed step count
    final_answer: str  # final answer
    error: str  # error message (if any)

# initialize state
initial_state = AgentState(
    messages=[],
    documents=[],
    step_count=0,
    final_answer="",
    error=""
)
```

## 5.3 @tool decorator

```python
from langchain_core.tools import tool
from typing import Literal

@tool
def get_weather(city: str) -> str:
    """get weather info for specified city"""
    # should call real API in actual project
    weather_data = {
        "北京": "晴朗，25°C",
        "上海": "多云，22°C",
        "广州": "小雨，28°C"
    }
    return weather_data.get(city, "未知城市")

@tool
def search_documents(query: str) -> list[str]:
    """search related documents"""
    # should call vector database in actual project
    return [
        f"文档1: {query} 相关内容 A",
        f"文档2: {query} 相关内容 B"
    ]

# tool list
tools = [get_weather, search_documents]
```

## 5.4 ToolNode：automatically handle tool calls

```python

# ToolNode 会自动：
# 1. parse tool_calls from LLM output
# 2. call corresponding tool function
# 3. wrap result as ToolMessage back to state

tool_node = ToolNode(tools)

# usage example
```

## 5.5 complete Agent graph

```python
from langgraph.graph import StateGraph, MessagesState, START, END
from langgraph.prebuilt import ToolNode
from typing import Annotated, operator

class AgentState(MessagesState):
    step_count: Annotated[int, operator.add]

def create_agent_graph():
    # 节点
    def llm_node(state: AgentState) -> dict:
        # simulate LLM call
        # 实际: from ai import createGateway, streamText
        #       gateway = createGateway({ apiKey: process.env.AI_GATEWAY_API_KEY })
        #       model = gateway('openai/gpt-4o')
        last_msg = state["messages"][-1]
        
        if last_msg.type == "human":
            return {
                "messages": [{
                    "role": "assistant",
                    "content": "",
                    "tool_calls": [{
                        "id": "call_1",
                        "name": "get_weather",
                        "args": {"city": "北京"}
                    }]
                }],
                "step_count": 1
            }
        return {"step_count": state.get("step_count", 0) + 1}
    
    # judge whether to continue calling tool
    def should_continue(state: AgentState) -> str:
        last_msg = state["messages"][-1]
        if hasattr(last_msg, 'tool_calls') and last_msg.tool_calls:
            return "tools"
        return "end"
    
    # build graph
    graph = StateGraph(AgentState)
    graph.add_node("llm", llm_node)
    graph.add_node("tools", ToolNode([get_weather, search_documents]))
    
    graph.add_edge(START, "llm")
    graph.add_conditional_edges("llm", should_continue, {"tools": "tools", "end": END})
    graph.add_edge("tools", "llm")
    
    return graph.compile()

# 执行
agent = create_agent_graph()
result = agent.invoke({
    "messages": [{"role": "user", "content": "北京天气怎么样？"}]
})
print(result["messages"][-1]["content"])
```

## 本章Summary

- `MessagesState` manage conversation message history
- 自定义stateuse `TypedDict` 扩展 `MessagesState`
- `@tool` decoratordefine tool，docstring as tool description
- `ToolNode` automatically handle tool call and result writeback
- conditional edge implements tool call loop logic
