## 5.1 MessagesState Deep Dive

`MessagesState` is LangGraph's built-in state type, containing `messages` field:

```python
from langgraph.graph import MessagesState
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage

# messages field structure
messages = [
    HumanMessage(content="Hello"),           # User message
    AIMessage(content="Hello! How can I help you?"),  # AI response
    # Inserted when tool call occurs
    AIMessage(
        content="",
        tool_calls=[{"id": "call_123", "name": "get_weather", "args": {"city": "Beijing"}}]
    ),
    ToolMessage(content="Beijing is sunny today, 25°C", tool_call_id="call_123")  # Tool result
]
```

## 5.2 Custom State

```python
from langgraph.graph import StateGraph, MessagesState, START, END
from typing import TypedDict, Annotated
import operator

class AgentState(MessagesState):
    # Use Annotated + operator.add to implement message append (non-overwrite)
    documents: Annotated[list, operator.add]  # Retrieved document list
    step_count: int  # Number of executed steps
    final_answer: str  # Final answer
    error: str  # Error message (if any)

# Initialize state
initial_state = AgentState(
    messages=[],
    documents=[],
    step_count=0,
    final_answer="",
    error=""
)
```

## 5.3 @tool Decorator

```python
from langchain_core.tools import tool
from typing import Literal

@tool
def get_weather(city: str) -> str:
    """Get weather information for specified city"""
    # In production, should call real API
    weather_data = {
        "Beijing": "Sunny, 25°C",
        "Shanghai": "Cloudy, 22°C",
        "Guangzhou": "Light rain, 28°C"
    }
    return weather_data.get(city, "Unknown city")

@tool
def search_documents(query: str) -> list[str]:
    """Search relevant documents"""
    # In production, should call vector database
    return [
        f"Document 1: {query} related content A",
        f"Document 2: {query} related content B"
    ]

# Tool list
tools = [get_weather, search_documents]
```

## 5.5 Complete Agent Graph

```python
from langgraph.graph import StateGraph, MessagesState, START, END
from langgraph.prebuilt import ToolNode
from typing import Annotated, operator

class AgentState(MessagesState):
    step_count: Annotated[int, operator.add]

def create_agent_graph():
    # Nodes
    def llm_node(state: AgentState) -> dict:
        # Simulate LLM call
        # Production: from ai import createGateway, streamText
        #            gateway = createGateway({ apiKey: process.env.AI_GATEWAY_API_KEY })
        #            model = gateway('openai/gpt-4o')
        last_msg = state["messages"][-1]

        if last_msg.type == "human":
            return {
                "messages": [{
                    "role": "assistant",
                    "content": "",
                    "tool_calls": [{
                        "id": "call_1",
                        "name": "get_weather",
                        "args": {"city": "Beijing"}
                    }]
                }],
                "step_count": 1
            }
        return {"step_count": state.get("step_count", 0) + 1}

    # Determine if tool call should continue
    def should_continue(state: AgentState) -> str:
        last_msg = state["messages"][-1]
        if hasattr(last_msg, 'tool_calls') and last_msg.tool_calls:
            return "tools"
        return "end"

    # Build graph
    graph = StateGraph(AgentState)
    graph.add_node("llm", llm_node)
    graph.add_node("tools", ToolNode([get_weather, search_documents]))

    graph.add_edge(START, "llm")
    graph.add_conditional_edges("llm", should_continue, {"tools": "tools", "end": END})
    graph.add_edge("tools", "llm")

    return graph.compile()

# Execute
agent = create_agent_graph()
result = agent.invoke({
    "messages": [{"role": "user", "content": "What's the weather in Beijing?"}]
})
print(result["messages"][-1]["content"])
```

## Chapter Summary

- `MessagesState` manages conversation message history
- Custom state extends `MessagesState` using `TypedDict`
- `@tool` decorator defines tools, docstring serves as tool description
- `ToolNode` automatically handles tool calling and result writeback
- Conditional edges implement tool call loop logic
