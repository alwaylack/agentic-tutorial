## 2.2 Redis primitive

Redis 最常message 端它核心操作只三个

| command | 作 | blocking版 |
|---|---|---|
| `RPUSH item` | producer | — |
| `LPUSH item` | | — |
| `RPOP ` | consumer | `BRPOP` |
| `LRANGE 0 -1` | view | — |

### 2.2.1 blockingBRPOP

`RPOP` 为空时会立 `nil`consumer断**忙 CPU**`BRPOP` blockingversion为空时待新

```python
import redis

r = redis.Redis(host='localhost', port=6379, db=0)

# producer：推入task
r.rpush('tasks', 'task_001')
r.rpush('tasks', 'task_002')

# consumer：blocking等待（最多等 5 秒）
item = r.brpop('tasks', timeout=5)
# item 形如 (b'tasks', b'task_001')
if item:
    queue_name, task_id = item
    print(f'Got task: {task_id.decode()}')
else:
    print('No tasks in queue')
```

```python
# 将task从 'high' queue移到 'low' queue（原子操作）
result = r.brpoplpush('high', 'low', timeout=5)
```

### 2.3.3 简单 Task

```python
# tasks.py
from app import app
import time

@app.task(bind=True, max_retries=3)
def call_llm(self, prompt: str, model: str = "gpt-4o-mini") -> dict:
    """模拟 LLM 调用（实际项目中替换为真实 API 调用）"""
    try:
        # 模拟 API 调用延迟
        time.sleep(2)
        return {
            "model": model,
            "response": f"这是针对 '{prompt}' 的模拟回复",
            "tokens": 42,
        }
    except Exception as exc:
        # 自动retry，指数退避
        raise self.retry(exc=exc, countdown=2 ** self.request.retries)
```

## 2.4 async模式LLM 调

中LLM 调通常**最慢环节**网络 + 把它放入让 Fast 立 ID通 WebSocket 获取

```python
# agent_tasks.py
from app import app
from openai import AsyncOpenAI
import asyncio

client = AsyncOpenAI(api_key="sk-...")

@app.task(bind=True, max_retries=5, retry_backoff=True)
def generate_agent_response(
    self,
    messages: list[dict],
    tools: list[dict] | None = None,
) -> dict:
    """
    Agent 核心推理task：
    - 将多轮对话送入 LLM
    - 解析 tool_calls 并execute（由 Worker 内部complete）
    - 返回final文本响应
    """
    try:
        # 调用 LLM（此处用 OpenAI SDK，可替换为任意 Provider）
        response = asyncio.run(
            client.chat.completions.create(
                model="gpt-4o",
                messages=messages,
                tools=tools or [],
            )
        )
        choice = response.choices[0]
        return {
            "role": choice.message.role,
            "content": choice.message.content,
            "tool_calls": [
                {
                    "id": tc.id,
                    "function": {
                        "name": tc.function.name,
                        "arguments": tc.function.arguments,
                    },
                }
                for tc in choice.message.tool_calls or []
            ],
        }
    except Exception as exc:
        # retry_backoff=True 时 Celery 自动指数退避retry
        raise self.retry(exc=exc)
```

### 2.4.1 Fast endpointsubmit并立

```python
# main.py
from fastapi import FastAPI, HTTPException
from agent_tasks import generate_agent_response

app = FastAPI(title="Agent asyncservice")

@app.post("/chat")
async def chat_endpoint(prompt: str, model: str = "gpt-4o"):
    """
    submit对话task，立即返回 task_id。
    客户端用 task_id pollingresult。
    """
    task = generate_agent_response.delay(
        messages=[{"role": "user", "content": prompt}],
        model=model,
    )
    return {"task_id": task.id, "status": "queued"}

@app.get("/chat/{task_id}")
async def get_result(task_id: str):
    """pollingtaskresult"""
    from celery.result import AsyncResult
    result = AsyncResult(task_id)

    if result.state == "PENDING":
        return {"status": "pending"}
    if result.state == "STARTED":
        return {"status": "processing"}
    if result.state == "FAILURE":
        raise HTTPException(status_code=500, detail=str(result.result))
    if result.state == "SUCCESS":
        return {"status": "done", "result": result.result}
    return {"status": result.state}
```

## 2.5 详解

生产环境中 LLM rate limiting429Celery

### 2.5.1 级别bind=True

```python
@app.task(bind=True, max_retries=5, retry_backoff=True)
def robust_llm_call(self, prompt: str):
    try:
        return do_api_call(prompt)
    except RateLimitError:
        # retry_backoff=True：第1次等1s，第2次等2s，第3次等4s...
        raise self.retry(exc=self.__class__.exc, countdown=2 ** self.request.retries)
    except ConnectionError:
        # ConnectionError 立即retry，不等
        raise self.retry(exc=self.__class__.exc, max_retries=10)
```

### 2.5.2 全局默认

```python
app.conf.update(
    # 所有 task 默认最多retry 3 次
    task_default_max_retries = 3,
    # 启用指数退避（需要 bind=True）
    task_default_retry_backoff = True,
    # 退避基准（秒）
    task_default_retry_backoff_base = 2,
    # task超时（硬超时，Worker 内强制 kill）
    task_soft_time_limit = 120,   # 软超时：引发 SoftTimeLimitExceeded
    task_time_limit = 180,        # 硬超时：直接 kill Worker 子进程
)
```
