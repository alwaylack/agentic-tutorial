```python
#!/usr/bin/env python3
"""统计输入文本的中文字数与总行数，用于周报交付前的自检。"""
import sys

text = sys.stdin.read() if not sys.argv[1:] else open(sys.argv[1], encoding="utf-8").read()
cjk = sum(1 for ch in text if "\u4e00" <= ch <= "\u9fff")
print(f"中文字符 {cjk} 个，共 {len(text.splitlines())} 行")
sys.exit(0 if 800 <= cjk <= 2000 else 1)
```

    name="周报助手",
    # 全书约定：OpenAI 兼容三方模型

## 20.5 Team  Skills： leader 

 `Skills`  `Team`  `skills` ，****，：

```python
import os
from pathlib import Path

from agno.agent import Agent
from agno.models.openai import OpenAIChat
from agno.skills import Skills, LocalSkills
from agno.team import Team

skills_dir = Path(__file__).parent / "my-skills"

implementer = Agent(
    name="Implementer",
    role="根据评审意见修改代码",
    model=OpenAIChat(id="deepseek-chat", api_key=os.getenv("DEEPSEEK_API_KEY"),
                     base_url="https://api.deepseek.com/v1"),
)

review_team = Team(
    name="Code Review Team",
    model=OpenAIChat(id="deepseek-chat", api_key=os.getenv("DEEPSEEK_API_KEY"),
                     base_url="https://api.deepseek.com/v1"),
    members=[implementer],
    skills=Skills(loaders=[LocalSkills(str(skills_dir))]),  # 队长持技能
    instructions=["先用你的技能完成代码评审，再把修改工作分派给 Implementer。"],
    markdown=True,
)

review_team.print_response("评审并改进这段代码：def calc(x,y): return x+y", stream=True)
```
